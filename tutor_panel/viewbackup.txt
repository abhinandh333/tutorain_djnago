import json
from datetime import datetime, timezone

from django.conf import settings
from django.contrib import messages
from django.core.mail import send_mail
from django.http import JsonResponse
from django.shortcuts import render, redirect
from django.views.decorators.csrf import csrf_protect, ensure_csrf_cookie
from django.views.decorators.http import require_POST

from firebase_admin import auth, get_app, firestore
from tutorain import firebase_config




@ensure_csrf_cookie
def login_page(request):
    if request.session.get("firebase_uid"):
        return redirect("tutor_panel:dashboard")

    return render(request, "tutor_panel/login.html")


@require_POST
@csrf_protect
def firebase_login(request):
    try:
        data = json.loads(request.body)
        id_token = data.get("idToken")

        if not id_token:
            return JsonResponse(
                {"success": False, "error": "Missing login token."},
                status=400,
            )

        decoded_token = auth.verify_id_token(id_token, app=get_app())

        uid = decoded_token["uid"]
        email = decoded_token.get("email", "").lower()

        # Only explicitly approved tutors can access the portal.
        approved_emails = [
            e.lower()
            for e in getattr(settings, "APPROVED_TUTOR_EMAILS", [])
        ]

        admin_emails = [
            e.lower()
            for e in getattr(settings, "ADMIN_EMAILS", [])
        ]

        if email not in approved_emails and email not in admin_emails:
            return JsonResponse(
                {
                    "success": False,
                    "error": "Your account is not approved. Please contact the administrator.",
                },
                status=403,
            )

        # Prevent session fixation.
        request.session.flush()
        request.session["firebase_uid"] = uid
        request.session["firebase_email"] = email

        if email in admin_emails:
            redirect_url = "/tutor_panel/admin/"
        else:
            redirect_url = "/tutor_panel/dashboard/"

        return JsonResponse({
            "success": True,
            "redirect_url": redirect_url,
        })

    except Exception as e:
        print("FIREBASE LOGIN ERROR:", repr(e))

        return JsonResponse(
            {
                "success": False,
                "error": "Login failed. Check the Django terminal for details."
            },
            status=401,
        )

    except Exception:
        return JsonResponse(
            {"success": False, "error": "Login failed. Please try again."},
            status=401,
        )


def dashboard(request):
    if not request.session.get("firebase_uid"):
        return redirect("tutor_panel:login")

    email = request.session.get("firebase_email", "").lower()
    approved_emails = [
        e.lower()
        for e in getattr(settings, "APPROVED_TUTOR_EMAILS", [])
    ]

    if email not in approved_emails:
        request.session.flush()
        return redirect("tutor_panel:login")

    return render(request, "tutor_panel/dashboard.html", {
        "tutor_email": email,
    })



def admin_dashboard(request):
    if not request.session.get("firebase_uid"):
        return redirect("tutor_panel:login")

    email = request.session.get("firebase_email", "").lower()

    admin_emails = [
        e.lower()
        for e in getattr(settings, "ADMIN_EMAILS", [])
    ]

    if email not in admin_emails:
        return redirect("tutor_panel:dashboard")

    return render(
        request,
        "tutor_panel/admin_dashboard.html",
        {
            "admin_email": email,
        },
    )


def logout_page(request):
    request.session.flush()
    return redirect("tutor_panel:login")

def manage_tutors(request):
    # Check that the user is logged in
    if not request.session.get("firebase_uid"):
        return redirect("tutor_panel:login")

    # Get the logged-in user's email
    user_email = request.session.get("firebase_email", "").lower()

    # Normalize the administrator email list
    admin_emails = [
        email.lower()
        for email in getattr(settings, "ADMIN_EMAILS", [])
    ]

    # Allow administrators only
    if user_email not in admin_emails:
        return redirect("tutor_panel:admin_dashboard")

    # Connect to Firestore
    db = firestore.client()

    # Save a new tutor
    if request.method == "POST":
        name = request.POST.get("name", "").strip()
        email = request.POST.get("email", "").strip().lower()
        phone = request.POST.get("phone", "").strip()
        subject = request.POST.get("subject", "").strip()

        # Validate the submitted details
        if not all([name, email, phone, subject]):
            messages.error(
                request,
                "Please fill in all required fields."
            )
            return redirect("tutor_panel:manage_tutors")

        # Use email as the Firestore document ID
        tutor_ref = db.collection("tutors").document(email)

        # Prevent duplicate tutor records
        if tutor_ref.get().exists:
            messages.error(
                request,
                "A tutor with this email already exists."
            )
            return redirect("tutor_panel:manage_tutors")

        # Save tutor details
        tutor_ref.set({
            "name": name,
            "email": email,
            "phone": phone,
            "subject": subject,
            "status": "Pending",
            "created_at": datetime.now(timezone.utc),
        })

        messages.success(
            request,
            f"Tutor {name} has been added successfully!"
        )

        return redirect("tutor_panel:manage_tutors")

    # Retrieve tutors from Firestore
    tutors = []

    for document in db.collection("tutors").stream():
        tutor = document.to_dict()
        tutors.append(tutor)

    # Sort tutors alphabetically
    tutors.sort(
        key=lambda tutor: tutor.get("name", "").lower()
    )

    return render(
        request,
        "tutor_panel/manage_tutors.html",
        {"tutors": tutors},
    )
